# LunaWindow data contract — v1.0.0

The single shared interface between the Python pipeline (`pipeline/`) and the web app (`web/`).
Change it only through the main loop / integrator; bump `contractVersion` on any breaking change.

## 1. Conventions (apply everywhere)

| Quantity | Convention |
|---|---|
| Body-fixed frame | Moon Mean-Earth / polar-axis (ME). Horizons reports the observer frame as `MEAN_ME (high precision)`; LOLA polar DEMs are ME. Any residual difference between the two realisations must be stated in `site.frameNote`. |
| Reference surface | Sphere, radius **1 737 400 m**. Latitude is planetocentric (= planetodetic on a sphere). |
| Longitude | East-positive degrees, `[0, 360)`. |
| Latitude | Degrees, north-positive (south pole = −90). |
| Heights / distances | **Metres** in every JSON field (Horizons takes km — convert at the request boundary only). |
| Angles | **Degrees** in every JSON field. |
| Azimuth | Clockwise from local north through east, `[0, 360)` — identical to Horizons `Azi_(a-app)`. |
| Elevation | Above the plane perpendicular to the local zenith (radial direction on the sphere) — identical to Horizons `Elev_(a-app)`. |
| Time | UTC, ISO-8601 with trailing `Z` (`2027-01-01T00:00:00Z`). Sample `i` is at `startUtc + i * stepSeconds`. Horizons `UT` time tags after 1962 are UTC. |
| Body position | Apparent direction of the body **centre** (Horizons quantity 4: light-time, aberration, light deflection; airless). |
| Missing value | JSON `null` (never NaN, never a sentinel number). |
| Number precision | az/el arrays rounded to 4 decimals; horizon elevations 4 decimals; distances to 1 m. |

## 2. Files

```
web/public/data/
  manifest.json            index, supported range, sources, disclaimers
  sites/<siteId>.json      one self-contained dataset per demonstration point
  map/<name>.png           polar-stereographic overview image (optional)
fixtures/analysis-golden.json   cross-language analysis test vectors (section 5)
```

Everything under `web/public/data/` and `fixtures/` is **generated by the pipeline**; nothing else writes there.
A dataset that is not real NASA-derived data must set `manifest.synthetic = true`; the UI then shows a permanent
"synthetic test data" banner. Synthetic data never lives in `web/public/data/`.

## 3. TypeScript shapes (normative)

```ts
// ---------- manifest.json ----------
interface Manifest {
  contractVersion: "1.0.0";
  synthetic: boolean;                 // false only for real Horizons/LOLA-derived data
  generatedAtUtc: string;
  generator: { name: string; version: string };
  supportedRange: { startUtc: string; endUtc: string; stepSeconds: number }; // endUtc = last sample time
  defaults: { siteId: string; durationHours: number; shiftHours: number };   // 168 and 24
  sites: SiteIndexEntry[];
  map: MapInfo | null;
  sources: SourceRef[];
  disclaimers: string[];              // shown verbatim in the assumptions panel
}

interface SiteIndexEntry {
  id: string;                         // kebab-case, stable
  name: string;                       // e.g. "Demo point A"
  locationHint: string;               // nearby named feature, descriptive only
  latDeg: number;
  lonEastDeg: number;
  file: string;                       // relative to the data dir, e.g. "sites/demo-a.json"
  hasTerrainHorizon: boolean;
}

interface MapInfo {
  image: string;                      // relative to the data dir
  widthPx: number;
  heightPx: number;
  projection: "south-polar-stereographic";
  referenceRadiusM: 1737400;
  // Projected metres. x = r*sin(lonEast), y = r*cos(lonEast), r = 2R*tan((90+lat)/2 deg).
  // Pixel: col = (x - xMinM)/(xMaxM - xMinM)*widthPx ; row = (yMaxM - y)/(yMaxM - yMinM)*heightPx
  xMinM: number; xMaxM: number; yMinM: number; yMaxM: number;
  rendering: string;                  // e.g. "hillshade of <DEM>, sun azimuth ..., illustrative only"
  sourceId: string;                   // -> SourceRef.id
}

interface SourceRef {
  id: string;
  title: string;
  url: string;
  usedFor: string;
  retrievedAtUtc: string | null;
  citation: string | null;
  sha256: string | null;              // of the downloaded/cached artefact when applicable
}

// ---------- sites/<siteId>.json ----------
interface SiteDataset {
  contractVersion: "1.0.0";
  synthetic: boolean;
  site: {
    id: string;
    name: string;
    locationHint: string;
    description: string;              // neutral; never "landing site approved/selected by NASA"
    latDeg: number;
    lonEastDeg: number;
    frame: string;                    // e.g. "Moon ME (Horizons MEAN_ME; LOLA ME)"
    frameNote: string;
    referenceRadiusM: 1737400;
    surfaceElevationM: number | null; // DEM-derived, relative to the reference sphere; null if no DEM
    observerHeightM: number;          // above the local surface
    observerAltitudeM: number;        // = (surfaceElevationM ?? 0) + observerHeightM ; sent to Horizons (as km)
    siteChecks: string[];             // what was verified against terrain data for this point
  };
  time: { startUtc: string; stepSeconds: number; count: number; scale: "UTC" };
  ephemeris: {
    sourceId: string;
    query: Record<string, string>;    // exact Horizons parameters (minus target COMMAND)
    retrievedAtUtc: string;
    corrections: string;              // "airless apparent, body centre, ..."
    bodies: { sun: BodyTrack; earth: BodyTrack };
  };
  horizons: {
    geometric: GeometricHorizon;      // always present
    terrain: TerrainHorizon | null;   // null when terrain processing is unavailable for the site
  };
}

interface BodyTrack {
  horizonsCommand: string;            // "10" (Sun) or "399" (Earth)
  azDeg: (number | null)[];           // length = time.count
  elDeg: (number | null)[];
  angularRadiusDeg: { min: number; mean: number; max: number }; // from Horizons Ang-diam / 2
  rawFile: string;                    // repo-relative path of the cached raw response
  rawSha256: string;
}

interface GeometricHorizon {
  kind: "geometric";
  id: "geometric-reference";
  label: string;                      // "Geometric reference horizon (no terrain)"
  elevationDeg: 0;                    // local horizontal plane
  uncertaintyDeg: 0;
  assumptions: string[];
}

interface TerrainHorizon {
  kind: "terrain";
  id: "terrain-lola";
  label: string;                      // "Terrain horizon from LOLA elevation data"
  azimuthStepDeg: number;             // bins start at azimuth 0; length*step = 360
  elevationDeg: number[];             // horizon elevation angle per azimuth bin
  distanceM: number[];                // ground distance to the horizon-defining terrain point
  coverageLimited: boolean[];         // true where the search ray left valid DEM coverage before searchExtentM
  searchExtentM: number;
  nearFieldStartM: number;            // first sampled distance from the observer
  curvatureModel: string;             // how lunar curvature was handled
  uncertaintyDeg: number;             // stated 1-sigma-ish horizon elevation uncertainty used for near-horizon flags
  dem: {
    sourceId: string;
    product: string;                  // file name
    resolutionM: number;
    projection: string;
    heightReference: string;          // "metres above 1737400 m sphere" (after scaling)
    scaling: string;                  // scale/offset/nodata handling actually applied
  };
  assumptions: string[];
}
```

## 4. Visibility analysis (normative algorithm)

Implemented twice — `pipeline/` (Python reference) and `web/src/analysis/` (TypeScript, what the UI runs) —
and held together by `fixtures/analysis-golden.json`.

**Inputs:** the site time grid, one `BodyTrack`, one horizon model, window start `T0` (must be a grid time),
`durationHours` (window end `T1 = T0 + duration`, also a grid time, inside the supported range), optional `limitHours`.

1. **Horizon lookup** `H(az)`: geometric → constant `elevationDeg`. Terrain → linear interpolation between the two
   neighbouring azimuth bins, wrapping 360 → 0.
2. **Margin** at sample `i`: `m_i = el_i − H(az_i)`; `null` if `az_i` or `el_i` is `null`.
3. **Sample state:** `visible` if `m_i > 0`, `blocked` if `m_i ≤ 0`, `unknown` if `null`.
4. **Segments** between consecutive samples `i, i+1` inside the window:
   - either sample unknown → the whole segment is `unknown`;
   - same state → that state;
   - different states → split at `t* = t_i + step * m_i / (m_i − m_{i+1})` (linear interpolation of the margin).
5. **Intervals:** merge adjacent pieces with equal state into maximal intervals covering `[T0, T1]` exactly.
6. **Near-horizon flag:** sample `i` is near-horizon when `|m_i| < band`,
   `band = body.angularRadiusDeg.max + horizon.uncertaintyDeg`. Maximal runs of flagged samples become
   `nearHorizonSpans` (`[t_first, t_last]`, clipped to the window). Meaning: the body's disc straddles the modelled
   horizon or the margin is inside the stated horizon uncertainty, so the visible/blocked label is not reliable there.
7. **Longest interruption:** the `blocked` interval with the greatest duration (earliest wins ties); `null` if none.
8. **Limit status** (only when `limitHours` is given), first match wins:
   - `exceeded` — some blocked interval has `durationHours > limitHours`;
   - `not-exceeded-truncated` — none exceeds, but a blocked interval touches `T0` or `T1`
     (its full duration may extend outside the analysed period);
   - `indeterminate-missing-data` — none exceeds, `unknownHours > 0`;
   - `within` — otherwise.

**Clarifications (both implementations must agree; covered by golden cases):**

- Zero-length pieces are dropped before merging. They arise only from an exact-zero margin next to a state change
  (margins `+1, 0, +1` give no blocked interval); two or more consecutive zero-margin samples do give one.
- Output timestamps are rounded to the nearest millisecond. Compare times and durations with
  `toleranceHours` (1e-6 h = 3.6 ms), never as exact strings.
- A `null` sample is never near-horizon and ends a near-horizon run. A single flagged sample gives a zero-length span.
- "Longer than the limit" means `durationHours > limitHours + 1e-9`, so floating-point noise cannot flip a verdict
  when a gap equals the limit.
- `GoldenCase.input.siteFile` is repo-relative (for example `web/public/data/sites/demo-a.json`).

```ts
type VisState = "visible" | "blocked" | "unknown";

interface VisInterval {
  state: VisState;
  startUtc: string;                   // ISO, millisecond precision
  endUtc: string;
  durationHours: number;
  touchesWindowStart: boolean;
  touchesWindowEnd: boolean;
  adjacentToUnknown: boolean;         // a neighbouring interval is "unknown"
}

interface AnalysisResult {
  body: "sun" | "earth";
  horizonId: "geometric-reference" | "terrain-lola";
  horizonKind: "geometric" | "terrain";
  windowStartUtc: string;
  windowEndUtc: string;
  stepSeconds: number;
  intervals: VisInterval[];
  nearHorizonSpans: { startUtc: string; endUtc: string }[];
  nearHorizonBandDeg: number;
  visibleHours: number;
  blockedHours: number;
  unknownHours: number;
  interruptionCount: number;          // number of blocked intervals
  longestInterruption: (VisInterval & { boundaryTruncated: boolean }) | null; // truncated = touches T0 or T1
  limit: null | {
    limitHours: number;
    status: "within" | "exceeded" | "not-exceeded-truncated" | "indeterminate-missing-data";
    exceedingIntervals: VisInterval[]; // blocked intervals longer than the limit
  };
}
```

Wording rules for anything derived from this result: gaps are "within the selected analysis interval";
a `boundaryTruncated` gap must be shown as possibly continuing outside it; limits are **user-defined tolerances**,
not computed battery or link budgets.

## 5. `fixtures/analysis-golden.json`

```ts
interface GoldenFile {
  contractVersion: "1.0.0";
  generatedBy: string;                // pipeline module + version
  toleranceHours: 1e-6;
  cases: GoldenCase[];
}
interface GoldenCase {
  name: string;
  // Inline synthetic input (small arrays) OR a reference to a real site dataset.
  input:
    | { kind: "inline"; time: SiteDataset["time"]; track: Pick<BodyTrack, "azDeg" | "elDeg" | "angularRadiusDeg">;
        horizon: GeometricHorizon | Pick<TerrainHorizon, "kind" | "id" | "azimuthStepDeg" | "elevationDeg" | "uncertaintyDeg"> }
    | { kind: "site"; siteFile: string; horizonId: "geometric-reference" | "terrain-lola" };
  body: "sun" | "earth";
  windowStartUtc: string;
  durationHours: number;
  limitHours: number | null;
  expected: AnalysisResult;
}
```

Required inline cases: always visible; always blocked (truncated both ends); one interior gap; gap touching window
start; gap touching window end; missing samples splitting a gap; terrain ridge with azimuth wrap-around;
exact-zero margin; limit exactly equal to the longest gap (`within`, strict `>`); +24 h shifted window of a case above.
Required site cases (once real data exists): at least one per body per horizon kind.
